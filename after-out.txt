[39m[31m<script>[39m[31m\n
    // KDP MART — Categories mega menu behaviour.\n
    //  • Click "Categories" toggles the panel (aria-expanded stays in sync).\n
    //  • Hovering / tabbing a main category swaps the subcategory column.\n
    //  • Mobile: tapping a category with children drills in (Back returns).\n
    //  • Closes on: toggle click, outside click, Escape, or any navigation.\n
    //  • Every panel is pre-rendered from the database, so hovering never\n
    //    triggers a request - no per-hover queries.\n
    (function () {\n
        'use strict';\n
\n
        var DESKTOP_QUERY = '(min-width: 992px)';\n
        var VIEWPORT_GAP = 14;\n
\n
        function isDesktop() {\n
            return window.matchMedia\n
                ? window.matchMedia(DESKTOP_QUERY).matches\n
                : window.innerWidth >= 992;\n
        }\n
\n
        function closest(element, selector) {\n
            var node = element;\n
            while (node && node.nodeType === 1) {\n
                if (node.matches && node.matches(selector)) { return node; }\n
                node = node.parentNode;\n
            }\n
            return null;\n
        }\n
\n
        function initMega(nav) {\n
            var toggle = nav.querySelector('.kdp-mega__toggle');\n
            var panel = nav.querySelector('.kdp-mega__panel');\n
            if (!toggle || !panel) { return; }\n
\n
            var catLinks = Array.prototype.slice.call(nav.querySelectorAll('.kdp-mega__cat'));\n
            var subPanels = Array.prototype.slice.call(nav.querySelectorAll('.kdp-mega__sub'));\n
\n
            function isOpen() {\n
                return nav.classList.contains('is-open');\n
            }\n
\n
            function activate(id) {\n
                if (id === null || typeof id === 'undefined') { return; }\n
\n
                catLinks.forEach(function (link) {\n
                    link.classList.toggle('is-active', link.getAttribute('data-kdp-cat') === String(id));\n
                });\n
                subPanels.forEach(function (section) {\n
                    section.classList.toggle('is-active', section.getAttribute('data-kdp-sub') === String(id));\n
                });\n
            }\n
\n
            // Keeps the absolutely positioned panel inside the viewport on both\n
            // axes: clamps the horizontal offset and caps the height so the\n
            // columns scroll internally instead of overflowing the page.\n
            function fit() {\n
                panel.style.removeProperty('--kdp-mega-max-h');\n
\n
                if (!isDesktop()) {\n
                    panel.style.left = '';\n
                    panel.style.right = '';\n
                    return;\n
                }\n
\n
                panel.style.left = '0';\n
                panel.style.right = 'auto';\n
\n
                var navRect = nav.getBoundingClientRect();\n
                var panelRect = panel.getBoundingClientRect();\n
                var minLeft = VIEWPORT_GAP - navRect.left;\n
                var maxLeft = (window.innerWidth - VIEWPORT_GAP - navRect.left) - panelRect.width;\n
                panel.style.left = Math.round(Math.min(0, Math.max(minLeft, maxLeft))) + 'px';\n
\n
                var top = panel.getBoundingClientRect().top;\n
                var maxHeight = Math.max(260, window.innerHeight - top - VIEWPORT_GAP);\n
                panel.style.setProperty('--kdp-mega-max-h', Math.round(maxHeight) + 'px');\n
            }\n
\n
            function onKeydown(event) {\n
                if (event.key === 'Escape' || event.key === 'Esc') {\n
                    event.preventDefault();\n
                    close(true);\n
                }\n
            }\n
\n
            function open() {\n
                nav.classList.add('is-open');\n
                toggle.setAttribute('aria-expanded', 'true');\n
                document.addEventListener('keydown', onKeydown);\n
                fit();\n
            }\n
\n
            function close(restoreFocus) {\n
                if (!isOpen()) { return; }\n
                nav.classList.remove('is-open');\n
                nav.classList.remove('is-drilled');\n
                toggle.setAttribute('aria-expanded', 'false');\n
                document.removeEventListener('keydown', onKeydown);\n
                if (restoreFocus) { toggle.focus(); }\n
            }\n
\n
            function visibleFocusable() {\n
                return Array.prototype.filter.call(\n
                    panel.querySelectorAll('a[href], button:not([disabled])'),\n
                    function (element) { return element.offsetParent !== null; }\n
                );\n
            }\n
\n
\n
            toggle.addEventListener('click', function (event) {\n
                event.preventDefault();\n
                if (isOpen()) { close(false); } else { open(); }\n
            });\n
\n
            // Hover (desktop) and keyboard focus both keep the right column in\n
            // sync with the highlighted category.\n
            nav.addEventListener('mouseover', function (event) {\n
                var link = closest(event.target, '.kdp-mega__cat');\n
                if (link && nav.contains(link)) { activate(link.getAttribute('data-kdp-cat')); }\n
            });\n
\n
            nav.addEventListener('focusin', function (event) {\n
                var link = closest(event.target, '.kdp-mega__cat');\n
                if (link) { activate(link.getAttribute('data-kdp-cat')); }\n
            });\n
\n
            nav.addEventListener('click', function (event) {\n
                var back = closest(event.target, '[data-kdp-back]');\n
                if (back) {\n
                    event.preventDefault();\n
                    nav.classList.remove('is-drilled');\n
                    var active = nav.querySelector('.kdp-mega__cat.is-active');\n
                    if (active) { active.focus(); }\n
                    return;\n
                }\n
\n
                var link = closest(event.target, '.kdp-mega__cat');\n
                if (!link) { return; }\n
